cmake_minimum_required(VERSION 3.26)
if(TARGET myproject::myproject)
    return()
endif()
project(myproject VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_CXX_SCAN_FOR_MODULES OFF)
if(PROJECT_IS_TOP_LEVEL AND NOT CMAKE_CONFIGURATION_TYPES AND NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release CACHE STRING "Build configuration" FORCE)
endif()
option(MYPROJECT_BUILD_PYTHON "Build the pybind11 extension" ${SKBUILD})
option(MYPROJECT_BUILD_TESTS "Build C++ tests" ${PROJECT_IS_TOP_LEVEL})
option(MYPROJECT_BUILD_EXAMPLES "Build the C++ example" ${PROJECT_IS_TOP_LEVEL})
option(MYPROJECT_INSTALL_SDK "Install the C++ SDK" ON)
option(MYPROJECT_BUNDLE_OPENMP "Bundle the shared SDK's OpenMP runtime" OFF)
if(MYPROJECT_BUNDLE_OPENMP AND NOT BUILD_SHARED_LIBS)
    message(FATAL_ERROR "Runtime bundling requires BUILD_SHARED_LIBS=ON")
endif()
include(GNUInstallDirs)
include(cmake/myprojectOpenMP.cmake)
find_package(OpenMP REQUIRED COMPONENTS CXX)
add_library(myproject src/distances.cpp)
add_library(myproject::myproject ALIAS myproject)
target_compile_features(myproject PUBLIC cxx_std_17)
target_include_directories(myproject PUBLIC
    $<BUILD_INTERFACE:${PROJECT_SOURCE_DIR}/include>
    $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
target_link_libraries(myproject PRIVATE OpenMP::OpenMP_CXX)
set_target_properties(myproject PROPERTIES POSITION_INDEPENDENT_CODE ON
    CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN ON
    VERSION ${PROJECT_VERSION} SOVERSION 0.1)
if(MSVC)
    target_compile_options(myproject PRIVATE /W4 /permissive- /utf-8)
else()
    target_compile_options(myproject PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(BUILD_SHARED_LIBS)
    target_compile_definitions(myproject PRIVATE MYPROJECT_EXPORTS PUBLIC MYPROJECT_SHARED)
endif()
if(APPLE)
    set_target_properties(myproject PROPERTIES INSTALL_RPATH "@loader_path")
elseif(UNIX)
    set_target_properties(myproject PROPERTIES INSTALL_RPATH "$ORIGIN")
endif()
if(MYPROJECT_BUILD_PYTHON)
    find_package(Python REQUIRED COMPONENTS Interpreter Development.Module)
    find_package(pybind11 3.0 CONFIG REQUIRED)
    pybind11_add_module(_core myproject/_core/bindings.cpp)
    target_link_libraries(_core PRIVATE myproject::myproject)
    if(APPLE)
        # The raw wheel must let delocate locate libomp's @rpath install name.
        get_filename_component(_omp_dir "${OpenMP_omp_LIBRARY}" DIRECTORY)
        set_target_properties(_core PROPERTIES INSTALL_RPATH "${_omp_dir}")
    endif()
    install(TARGETS _core LIBRARY DESTINATION myproject RUNTIME DESTINATION myproject)
endif()
if(MYPROJECT_INSTALL_SDK)
    install(TARGETS myproject EXPORT myprojectTargets
        ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR}
        LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR}
        RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
    install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
    install(FILES LICENSE DESTINATION ${CMAKE_INSTALL_DATADIR}/licenses/myproject)
    include(CMakePackageConfigHelpers)
    configure_package_config_file(cmake/myprojectConfig.cmake.in
        ${PROJECT_BINARY_DIR}/myprojectConfig.cmake
        INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/myproject)
    write_basic_package_version_file(${PROJECT_BINARY_DIR}/myprojectConfigVersion.cmake
        VERSION ${PROJECT_VERSION} COMPATIBILITY SameMinorVersion)
    install(EXPORT myprojectTargets NAMESPACE myproject::
        DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/myproject)
    install(FILES ${PROJECT_BINARY_DIR}/myprojectConfig.cmake
        ${PROJECT_BINARY_DIR}/myprojectConfigVersion.cmake
        cmake/myprojectOpenMP.cmake
        DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/myproject)
    install(DIRECTORY licenses/ DESTINATION ${CMAKE_INSTALL_DATADIR}/licenses/myproject)
    if(MYPROJECT_BUNDLE_OPENMP)
        file(GLOB _omp_redist "$ENV{VCToolsRedistDir}/x64/Microsoft.VC*.OpenMP")
        configure_file(cmake/BundleOpenMP.cmake.in ${PROJECT_BINARY_DIR}/BundleOpenMP.cmake @ONLY)
        install(SCRIPT ${PROJECT_BINARY_DIR}/BundleOpenMP.cmake)
    endif()
endif()
if(MYPROJECT_BUILD_EXAMPLES)
    add_executable(myproject_example examples/cpp/main.cpp)
    target_link_libraries(myproject_example PRIVATE myproject::myproject)
endif()
if(MYPROJECT_BUILD_TESTS)
    enable_testing()
    add_executable(test_distances tests/test_distances.cpp)
    target_link_libraries(test_distances PRIVATE myproject::myproject)
    add_test(NAME distances COMMAND test_distances)
    set_tests_properties(distances PROPERTIES ENVIRONMENT "OMP_NUM_THREADS=2;OMP_DYNAMIC=FALSE")
endif()
